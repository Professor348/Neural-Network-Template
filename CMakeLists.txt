cmake_minimum_required(VERSION 3.16)
project(NeuralTemplate VERSION 1.0.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
option(NN_BUILD_EXAMPLES "Build separately linked synthetic demo programs" ON)
option(NN_BUILD_BENCHMARKS "Build synthetic performance benchmark (no timing assertions)" OFF)
option(NN_ENABLE_GPU "Enable real OpenCL 1.2 float64 GPU training and inference" OFF)
option(NN_ENABLE_NPU "Enable OpenVINO Intel NPU inference (explicit reduced-precision opt-in)" OFF)
add_library(nn src/Network.cpp src/Training.cpp src/Validation.cpp
    src/numerics/Functions.cpp src/devices/Selection.cpp src/backends/Cpu.cpp src/backends/Execution.cpp)
add_library(nn::nn ALIAS nn)
if(NN_ENABLE_GPU)
    # OpenCL SDK types are private; pinned Khronos headers are bundled for compilation.
    if(NOT TARGET OpenCL::OpenCL)
        find_library(NN_OPENCL_LIBRARY NAMES OpenCL)
        if(NOT NN_OPENCL_LIBRARY)
            message(FATAL_ERROR "NN_ENABLE_GPU needs the OpenCL ICD loader/development library (OpenCL).")
        endif()
        add_library(OpenCL::OpenCL UNKNOWN IMPORTED)
        set_target_properties(OpenCL::OpenCL PROPERTIES IMPORTED_LOCATION "${NN_OPENCL_LIBRARY}")
    endif()
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/backends/opencl/kernels.cl" NN_OPENCL_SOURCE)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS src/backends/opencl/kernels.cl)
    configure_file(src/backends/opencl/KernelSource.h.in generated/KernelSource.h @ONLY)
    target_sources(nn PRIVATE src/backends/opencl/OpenClSession.cpp src/backends/opencl/Discovery.cpp src/backends/opencl/OpenClRuntime.cpp)
    target_include_directories(nn PRIVATE third_party/OpenCL "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_compile_definitions(nn PRIVATE NN_ENABLE_GPU=1)
    target_link_libraries(nn PRIVATE OpenCL::OpenCL)
endif()
if(NN_ENABLE_NPU)
    # OpenVINO remains private to optional implementation files; consumers find its runtime transitively.
    if(NOT TARGET openvino::runtime)
        find_package(OpenVINO 2024.6 REQUIRED COMPONENTS Runtime)
    endif()
    target_compile_definitions(nn PRIVATE NN_ENABLE_NPU=1)
    target_link_libraries(nn PRIVATE openvino::runtime)
    target_sources(nn PRIVATE src/backends/openvino/NpuSession.cpp
        src/backends/openvino/NpuRuntime.cpp src/backends/openvino/NpuState.cpp src/backends/openvino/Graph.cpp
        src/backends/openvino/Snapshot.cpp)
endif()
target_compile_features(nn PUBLIC cxx_std_17)
target_include_directories(nn PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}> PRIVATE src)
if(MSVC)
    target_compile_options(nn PRIVATE /W4)
else()
    target_compile_options(nn PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(NN_BUILD_EXAMPLES)
    # Demo names map directly to independently linked public-API applications.
    foreach(demo_name xor regression classification devices npu_inference)
        add_executable(nn_${demo_name} examples/${demo_name}.cpp)
        target_link_libraries(nn_${demo_name} PRIVATE nn::nn)
    endforeach()
endif()
if(NN_BUILD_BENCHMARKS)
    add_executable(nn_benchmark tests/cpu_performance.cpp)
    target_link_libraries(nn_benchmark PRIVATE nn::nn)
    add_executable(nn_gpu_benchmark tests/gpu_performance.cpp)
    target_link_libraries(nn_gpu_benchmark PRIVATE nn::nn)
endif()
if(BUILD_TESTING)
    if(NN_BUILD_EXAMPLES)
        # Each demo is a smoke test; NPU absence is an explicit skip, not a CPU substitute.
        foreach(demo_name xor regression classification devices npu_inference)
            add_test(NAME example_${demo_name} COMMAND nn_${demo_name})
            set_tests_properties(example_${demo_name} PROPERTIES LABELS "example" TIMEOUT 120)
        endforeach()
        set_tests_properties(example_npu_inference PROPERTIES SKIP_RETURN_CODE 77)
    endif()
    foreach(test_name numerics learning validation devices kernel_reference parallel_kernels cpu_regression npu_policy)
        add_executable(nn_test_${test_name} tests/${test_name}.cpp)
        target_link_libraries(nn_test_${test_name} PRIVATE nn::nn)
        target_include_directories(nn_test_${test_name} PRIVATE src)
        add_test(NAME ${test_name} COMMAND nn_test_${test_name})
    endforeach()
    # Test-only OpenCL shim executes real session code without linking the physical OpenCL loader.
    add_executable(nn_test_opencl_session_mock tests/opencl_session_mock.cpp
        src/backends/opencl/OpenClSession.cpp src/backends/opencl/OpenClRuntime.cpp
        src/backends/Cpu.cpp src/numerics/Functions.cpp)
    target_compile_features(nn_test_opencl_session_mock PRIVATE cxx_std_17)
    target_include_directories(nn_test_opencl_session_mock PRIVATE include src third_party/OpenCL)
    add_test(NAME opencl_session_mock COMMAND nn_test_opencl_session_mock)
    if(NN_ENABLE_NPU)
        # Actual OpenVINO graph semantics tested on CPU; never reported as NPU hardware validation.
        add_executable(nn_test_npu_graph tests/npu_graph.cpp
            src/backends/openvino/Graph.cpp src/backends/openvino/Snapshot.cpp)
        target_compile_features(nn_test_npu_graph PRIVATE cxx_std_17)
        target_include_directories(nn_test_npu_graph PRIVATE include src)
        target_link_libraries(nn_test_npu_graph PRIVATE openvino::runtime)
        add_test(NAME npu_graph_cpu_validation COMMAND nn_test_npu_graph)
        set_tests_properties(npu_graph_cpu_validation PROPERTIES SKIP_RETURN_CODE 77)
    endif()
    # SDK-independent exact snapshot and reduced-precision validation.
    add_executable(nn_test_npu_snapshot tests/npu_snapshot.cpp src/backends/openvino/Snapshot.cpp)
    target_compile_features(nn_test_npu_snapshot PRIVATE cxx_std_17)
    target_include_directories(nn_test_npu_snapshot PRIVATE include src)
    add_test(NAME npu_snapshot COMMAND nn_test_npu_snapshot)
    # Isolated NPU boundary mock: exercises real dispatch without any SDK or physical NPU.
    add_executable(nn_test_npu_dispatch_mock tests/npu_dispatch_mock.cpp
        src/backends/Execution.cpp src/devices/Selection.cpp src/backends/Cpu.cpp src/numerics/Functions.cpp)
    target_compile_features(nn_test_npu_dispatch_mock PRIVATE cxx_std_17)
    target_compile_definitions(nn_test_npu_dispatch_mock PRIVATE NN_ENABLE_NPU=1)
    target_include_directories(nn_test_npu_dispatch_mock PRIVATE include src)
    add_test(NAME npu_dispatch_mock COMMAND nn_test_npu_dispatch_mock)
    add_executable(nn_test_hardware tests/hardware.cpp)
    target_link_libraries(nn_test_hardware PRIVATE nn::nn)
    add_test(NAME accelerator_execution COMMAND nn_test_hardware)
    set_tests_properties(accelerator_execution PROPERTIES SKIP_RETURN_CODE 77)
    add_executable(nn_test_npu_hardware tests/npu_hardware.cpp)
    target_link_libraries(nn_test_npu_hardware PRIVATE nn::nn)
    # These cases require physical NPUs; serialize the shared device's compilation/cache workload.
    foreach(npu_case inference loss synchronization contracts)
        add_test(NAME npu_hardware_${npu_case} COMMAND nn_test_npu_hardware ${npu_case})
        set_tests_properties(npu_hardware_${npu_case} PROPERTIES SKIP_RETURN_CODE 77
            LABELS "npu;hardware" RESOURCE_LOCK nn_npu_hardware TIMEOUT 180)
    endforeach()
    set_tests_properties(npu_policy npu_snapshot npu_dispatch_mock PROPERTIES LABELS "npu;software")
    if(NN_ENABLE_NPU)
        set_tests_properties(npu_graph_cpu_validation PROPERTIES LABELS "npu;graph;cpu")
    endif()
endif()
install(TARGETS nn EXPORT nnTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT nnTargets NAMESPACE nn:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/nn)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/nnConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/nnConfig.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/nn)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/nnConfigVersion.cmake
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/nnConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/nnConfigVersion.cmake
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/nn)
